# UNLOADED leader-editor candidate. GUI Apply stays disabled until transport proof.
# COUNTRY CURRENT/ROOT must be the local player's player_context (QA: own POL).
# Dependency: main mod RN_B_0..31 scripted localisation + rn_inline_payload.txt.
# Never opens rn_input or changes rn_ttype/rn_tchar. No native transport is wired.
# Future command inputs: fresh34 rn_native_* records + rn_ip_cmd_epoch (15 ASCII).
# Epoch is captured with the same fresh native field snapshot, not at later dispatch.
# Ruling-leader selection has bounded static proof; parser/temp_var and Apply need runtime QA.
# Error: 0 delegated rename, 1/4/5/6 payload, 7 context/open, 8 epoch, 9 target.

rn_inline_leader_clear_target = {
	clear_array = rn_ip_leader_target
	clear_variable = rn_ip_leader_identity
}

rn_inline_leader_refresh = {
	clear_array = rn_ip_leader_owned
	clear_array = rn_ip_leader_current
	every_character = {
		ROOT = { add_to_array = { rn_ip_leader_owned = PREV } }
		if = {
			limit = { is_country_leader = yes }
			ROOT = { add_to_array = { rn_ip_leader_current = PREV } }
		}
	}
}

rn_inline_leader_open = {
	if = {
		limit = {
			tag = POL
			is_ai = no
			NOT = { has_country_flag = rn_input }
			NOT = { has_country_flag = rn_ip_leader_open }
			NOT = { has_country_flag = rn_ip_country_open }
		}
		rn_inline_leader_clear_target = yes
		rn_inline_leader_refresh = yes
		# Reject ambiguity instead of choosing the last country-leader role in a loop.
		if = {
			limit = { check_variable = { rn_ip_leader_current^num = 1 } }
			set_variable = { rn_ip_leader_identity = rn_ip_leader_current^0 }
			add_to_array = { rn_ip_leader_target = rn_ip_leader_identity }
			clr_country_flag = rn_ip_party_neutrality_open
			add_to_variable = { rn_ip_epoch = 1 }
			set_country_flag = rn_ip_leader_open
		}
		else = { log = "RN inline leader Open rejected: current candidates [?rn_ip_leader_current^num]" }
	}
}

rn_inline_leader_cancel = {
	if = {
		limit = { tag = POL is_ai = no has_country_flag = rn_ip_leader_open }
		clr_country_flag = rn_ip_leader_open
		rn_inline_leader_clear_target = yes
		add_to_variable = { rn_ip_epoch = 1 }
	}
}

rn_inline_leader_apply = {
	set_temp_variable = { rn_ip_leader_error = 7 }
	if = {
		limit = {
			tag = POL
			is_ai = no
			has_country_flag = rn_ip_leader_open
			NOT = { has_country_flag = rn_ip_party_neutrality_open }
			NOT = { has_country_flag = rn_ip_country_open }
			NOT = { has_country_flag = rn_input }
		}
		set_temp_variable = { rn_ip_leader_error = 8 }
		if = {
			limit = {
				check_variable = { rn_ip_epoch > 0 }
				check_variable = { temp_var:rn_ip_cmd_epoch = rn_ip_epoch }
			}
			set_temp_variable = { rn_ip_leader_error = 9 }
			rn_inline_leader_refresh = yes
			if = {
				limit = {
					has_variable = rn_ip_leader_identity
					check_variable = { rn_ip_leader_target^num = 1 }
					check_variable = { rn_ip_leader_current^num = 1 }
					is_in_array = { array = rn_ip_leader_target value = rn_ip_leader_identity }
				}
				for_each_scope_loop = {
					array = rn_ip_leader_target
					if = {
						limit = {
							is_country_leader = yes
							is_in_array = { array = ROOT.rn_ip_leader_owned value = THIS }
							is_in_array = { array = ROOT.rn_ip_leader_current value = THIS }
						}
						# All target/epoch guards pass before the first buffer write.
						ROOT = {
							rn_inline_copy_payload = yes
							set_temp_variable = { rn_ip_leader_error = temp_var:rn_inline_error }
						}
						if = {
							limit = { check_variable = { temp_var:rn_ip_leader_error = 0 } }
							meta_effect = {
								text = { set_character_name = "[N]" }
								N = "[ROOT.RN_B_0][ROOT.RN_B_1][ROOT.RN_B_2][ROOT.RN_B_3][ROOT.RN_B_4][ROOT.RN_B_5][ROOT.RN_B_6][ROOT.RN_B_7][ROOT.RN_B_8][ROOT.RN_B_9][ROOT.RN_B_10][ROOT.RN_B_11][ROOT.RN_B_12][ROOT.RN_B_13][ROOT.RN_B_14][ROOT.RN_B_15][ROOT.RN_B_16][ROOT.RN_B_17][ROOT.RN_B_18][ROOT.RN_B_19][ROOT.RN_B_20][ROOT.RN_B_21][ROOT.RN_B_22][ROOT.RN_B_23][ROOT.RN_B_24][ROOT.RN_B_25][ROOT.RN_B_26][ROOT.RN_B_27][ROOT.RN_B_28][ROOT.RN_B_29][ROOT.RN_B_30][ROOT.RN_B_31]"
							}
						}
					}
				}
			}
		}
	}
	if = {
		limit = { check_variable = { temp_var:rn_ip_leader_error = 0 } }
		# Do not mutate an array while its scope iterator is active.
		rn_inline_leader_cancel = yes
		clear_array = rn_buf
		set_variable = { rn_len = 0 }
	}
	else = { log = "RN inline leader Apply rejected: error [?temp_var:rn_ip_leader_error]" }
}
