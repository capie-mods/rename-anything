rn_pqa_snapshot = {
	clear_array = rn_pqa_before_buf
	set_variable = { rn_pqa_before_len = rn_len }
	set_variable = { rn_pqa_had_len = 0 }
	if = { limit = { has_variable = rn_len } set_variable = { rn_pqa_had_len = 1 } }
	for_each_loop = {
		array = rn_buf
		value = rn_pqa_v
		add_to_array = { rn_pqa_before_buf = rn_pqa_v }
	}
}

rn_pqa_compare_snapshot = {
	set_temp_variable = { rn_pqa_diff = 0 }
	if = {
		limit = { NOT = { check_variable = { rn_len = rn_pqa_before_len } } }
		add_to_temp_variable = { rn_pqa_diff = 1 }
	}
	if = {
		limit = { NOT = { check_variable = { rn_buf^num = rn_pqa_before_buf^num } } }
		add_to_temp_variable = { rn_pqa_diff = 1 }
	}
	if = {
		limit = { check_variable = { rn_pqa_had_len = 0 } has_variable = rn_len }
		add_to_temp_variable = { rn_pqa_diff = 1 }
	}
	if = {
		limit = { check_variable = { rn_pqa_had_len = 1 } NOT = { has_variable = rn_len } }
		add_to_temp_variable = { rn_pqa_diff = 1 }
	}
	for_each_loop = {
		array = rn_buf
		value = rn_pqa_v
		index = rn_pqa_i
		if = {
			limit = { NOT = { check_variable = { rn_pqa_v = rn_pqa_before_buf^rn_pqa_i } } }
			add_to_temp_variable = { rn_pqa_diff = 1 }
		}
	}
}

rn_pqa_restore_snapshot = {
	clear_array = rn_buf
	for_each_loop = {
		array = rn_pqa_before_buf
		value = rn_pqa_v
		add_to_array = { rn_buf = rn_pqa_v }
	}
	if = {
		limit = { check_variable = { rn_pqa_had_len = 1 } }
		set_variable = { rn_len = rn_pqa_before_len }
	}
	else = { clear_variable = rn_len }
	rn_pqa_compare_snapshot = yes
	if = {
		limit = { check_variable = { rn_pqa_diff = 0 } }
		log = "RN_PQA restored PASS"
	}
	else = { log = "RN_PQA restored FAIL diff=[?rn_pqa_diff|0]" }
}

rn_pqa_finish_snapshot = {
	rn_pqa_compare_snapshot = yes
	# Roll back even a failed diagnostic; do not silently report it as unchanged.
	if = { limit = { check_variable = { rn_pqa_diff > 0 } } rn_pqa_restore_snapshot = yes }
	clear_array = rn_pqa_before_buf
	clear_variable = rn_pqa_before_len
	clear_variable = rn_pqa_had_len
}

rn_pqa_check_validation = {
	rn_inline_validate_payload = yes
	rn_pqa_compare_snapshot = yes
	if = {
		limit = {
			check_variable = { temp_var:rn_inline_error = rn_pqa_expected }
			check_variable = { temp_var:rn_inline_bad_index = rn_pqa_expected_bad }
			check_variable = { rn_pqa_diff = 0 }
		}
		log = "RN_PQA case=[?rn_pqa_case|0] PASS err=[?rn_inline_error|0] bad=[?rn_inline_bad_index|0] unchanged=1"
	}
	else = { log = "RN_PQA case=[?rn_pqa_case|0] FAIL err=[?rn_inline_error|0] bad=[?rn_inline_bad_index|0] diff=[?rn_pqa_diff|0]" }
}
