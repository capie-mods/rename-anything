# 0x546530; retained typed-state-variable-decompile.txt:436-444.
# Explicit reads request the command temp namespace, not country variables.
# Freshness itself is a native producer/lifetime requirement, not proved here:
# replace reserved keys rather than appending duplicate or stale records.
#
# Scratch tempvars, not extra inputs: rn_native_probe_error/index/bad_index/char/remainder.
# Error 0 means delegated to EXISTING Apply, not proof that a name was changed.
# 1 missing/false ok; 2 no rn_input; 3 unknown target type; 4 invalid length;
# 5 invalid active code; 6 nonzero padding. bad_index=-1 unless a slot failed.
# Rejection changes only scratch tempvars/logs, never rn_buf/rn_len/targets/flags.
#
# CHARS/MAXLEN source: tools/gen.py:35-65 (84 consecutive logical codes, max 32).
# Generic syntax primary: documentation/effects_documentation.md
# add_to_array:2278-2291, modulo_temp_variable:4986-4997,

# Internal preflight helper. Caller assigns char and starts index at 0.
# It never mutates persistent text or target state.
rn_native_probe_validate_slot = {
	if = {
		limit = { check_variable = { temp_var:rn_native_probe_error = 0 } }
		if = {
			limit = { check_variable = { temp_var:rn_native_probe_index < temp_var:rn_native_len } }
			set_temp_variable = { rn_native_probe_remainder = temp_var:rn_native_probe_char }
			modulo_temp_variable = { var = rn_native_probe_remainder value = 1 }
			if = {
				limit = {
					NOT = {
						AND = {
							check_variable = { temp_var:rn_native_probe_char > 0 }
							check_variable = { temp_var:rn_native_probe_char < 85 }
							check_variable = { temp_var:rn_native_probe_remainder = 0 }
						}
					}
				}
				set_temp_variable = { rn_native_probe_error = 5 }
				set_temp_variable = { rn_native_probe_bad_index = temp_var:rn_native_probe_index }
			}
		}
		else_if = {
			limit = { NOT = { check_variable = { temp_var:rn_native_probe_char = 0 } } }
			set_temp_variable = { rn_native_probe_error = 6 }
			set_temp_variable = { rn_native_probe_bad_index = temp_var:rn_native_probe_index }
		}
	}
	add_to_temp_variable = { rn_native_probe_index = 1 }
}
