# Rename Anything in-place editor (iteration 3, M3): UI-only status of the last REJECTED inline Apply.
# Called by scripted GUI callbacks only, strictly AFTER the guarded Apply helper (status_record)
# or after Open/Cancel (status_clear). These variables are display-only: no guard, trigger,
# validator, copy or setter reads them, and they never touch names, buffers, pins, flags or epoch.
# Codes: 1 native ok=0 (empty, more than 32 characters or a character outside the alphabet),
# 4 length, 5 character code, 6 padding; 7..10 guard (editor closed/stale epoch/target changed);
# 12 = no command epoch from the native editor bridge (extension not running, DLL-less mode).

rn_ip_status_clear = {
	clear_variable = rn_ip_status_err
	clear_variable = rn_ip_status_field
}

# Rejected Apply without the bridge's command epoch: the extension is not running (INT-M1).
rn_ip_status_no_bridge = {
	if = {
		limit = { has_variable = rn_ip_status_err check_variable = { temp_var:rn_ip_cmd_epoch = 0 } }
		set_variable = { rn_ip_status_err = 12 }
	}
}

# Pair editors (country, party): temps rn_pair_err / rn_pair_field from the guarded helper.
rn_ip_status_record_pair = {
	if = {
		limit = { check_variable = { temp_var:rn_pair_err > 0 } }
		set_variable = { rn_ip_status_err = temp_var:rn_pair_err }
		set_variable = { rn_ip_status_field = temp_var:rn_pair_field }
		rn_ip_status_no_bridge = yes
	}
	else = { rn_ip_status_clear = yes }
}

# Leader editor: temp rn_ip_leader_error from the guarded helper; single field.
rn_ip_status_record_leader = {
	if = {
		limit = { check_variable = { temp_var:rn_ip_leader_error > 0 } }
		set_variable = { rn_ip_status_err = temp_var:rn_ip_leader_error }
		set_variable = { rn_ip_status_field = 1 }
		rn_ip_status_no_bridge = yes
	}
	else = { rn_ip_status_clear = yes }
}
