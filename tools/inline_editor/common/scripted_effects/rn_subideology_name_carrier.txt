# Removes the name; the game's name returns. Scope: the country.
rn_subideology_name_carrier_clear = {
	clear_array = rn_sn_leader
	clear_array = capie_sn_n
	clear_variable = capie_sn_ver
	rn_subideology_name_commit = yes
}

# Scope: the country. The commit: a new revision (previous + 1; a missing one reads as 0).
# Always the last statement of a change.
rn_subideology_name_commit = {
	set_temp_variable = { rn_sn_rev = capie_sn_rev }
	add_to_temp_variable = { rn_sn_rev = 1 }
	set_variable = { capie_sn_rev = temp_var:rn_sn_rev }
}
