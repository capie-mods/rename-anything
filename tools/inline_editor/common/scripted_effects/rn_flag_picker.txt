# Flag picker (ADR-0032, decisions "2026-10-14 - flagi"; plugin/capie/CONTRACT.md section 17).
# The windows are generated (tools/modgen/features/flag_picker.py): capie_v1_ed__rn__flag for the
# player's own country (politics view) and capie_v1_ed__rn__subflag for another country (diplomacy
# view, rights of the rn_edit_rights rule as for names). The catalog lookups they call are generated
# too (rn_flag_catalog.txt: rn_fp_catalog, rn_fp_group_of, rn_fp_fill_group, rn_fp_fill_letter,
# rn_fp_resolve_game, rn_fp_write_stem, rn_fp_cos_write, rn_fp_cos_match).
#
# State (on ROOT, the player; nothing here is game state of another country):
#   flags rn_fp_open_own / rn_fp_open_sub (one picker open), rn_fp_view_target (the pool shown is
#   the target's own), rn_fp_view_other (the "Other" pool is shown);
#   rn_fp_target + rn_fp_target_arr  the country whose flag changes, pinned at opening;
#   rn_fp_epoch                      CAPIE opening epoch (carrier capie_v1_ep__rn__<editor>);
#   rn_fp_view                       1 flags of one pool, 2 country list (rule "all flags");
#   rn_fp_src + rn_fp_src_arr        country whose pool is shown;
#   rn_fp_letter                     letter of the country list (1..26), 27 = "Other";
#   rn_fp_page / _pages / _page_disp paging (ten per page); rn_fp_list, rn_fp_cells, rn_fp_rows;
#   rn_fp_sel, rn_fp_init, rn_fp_ovl selected index, the index shown at opening, overlord's index;
#   rn_fp_dirty                      refresh gate of the own picker (dirty = rn_fp_dirty).
# On the country whose flag changes: the CAPIE layer capie_cf_* (section 17.2), the mod's own
# rn_cf_idx (catalog index) and rn_cf_cat (catalog fingerprint), and the flag cooldown
# rn_ip_cd_flag_until / rn_ip_cd_flag_date (its own counter, the same length as the name's).
#
# Errors (temp rn_cv1_err, shown through rn_cv1_st_err): 10 CAPIE missing, 11 cooldown, 12 stale
# opening, 13 editor unavailable, 15 CAPIE older than API level 6, 16 CAPIE without the flag layer
# (capie_flag is not 1), 17 flag not in the pool of the current game rule, 20 not allowed.

rn_fp_touch = {
	add_to_variable = { rn_fp_dirty = 1 }
	if = { limit = { check_variable = { rn_fp_dirty > 999999 } } set_variable = { rn_fp_dirty = 1 } }
}

# ----------------------------------------------------------------------------- opening, closing

# Own country (politics view). Scope: the player.
rn_fp_open_own = {
	if = {
		limit = {
			tag = ROOT
			is_ai = no
			NOT = { has_country_flag = rn_open }
			NOT = { has_country_flag = rn_input }
			NOT = { has_country_flag = rn_ip_leader_open }
			NOT = { has_country_flag = rn_ip_party_neutrality_open }
			NOT = { rn_feature_flag_off = yes }
		}
		rn_inline_pair_cancel_country = yes
		rn_pp_close = yes
		rn_fp_close = yes
		clear_array = rn_fp_target_arr
		add_to_array = { rn_fp_target_arr = THIS }
		set_variable = { rn_fp_target = THIS }
		set_country_flag = rn_fp_open_own
		rn_fp_open_common = yes
	}
}

# Another country (diplomacy view). Scope: that country; ROOT is the player.
rn_fp_open_sub = {
	if = {
		limit = {
			NOT = { rn_feature_flag_off = yes }
			NOT = { rn_feature_subjects_off = yes }
			rn_sub_may_edit = yes
			ROOT = { is_ai = no NOT = { has_country_flag = rn_open } NOT = { has_country_flag = rn_input } }
		}
		ROOT = {
			rn_sub_close = yes
			rn_fac_close = yes
			rn_pp_close = yes
			rn_fp_close = yes
			clear_array = rn_fp_target_arr
			add_to_array = { rn_fp_target_arr = PREV }
			set_variable = { rn_fp_target = PREV }
			set_country_flag = rn_fp_open_sub
			rn_fp_open_common = yes
		}
	}
}

# ROOT scope, target pinned. The overlord's flag is resolved now (a snapshot, ADR-0032) and only
# when the rule offers it; the current flag is preselected so the preview is never empty.
rn_fp_open_common = {
	add_to_variable = { rn_fp_epoch = 1 }
	if = { limit = { check_variable = { rn_fp_epoch > 999999 } } set_variable = { rn_fp_epoch = 1 } }
	rn_cv1_status_clear = yes
	set_variable = { rn_fp_ovl = 0 }
	if = {
		limit = {
			NOT = { has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_OWN } }
			var:rn_fp_target = { is_subject = yes }
		}
		var:rn_fp_target = { overlord = { rn_fp_current = yes } }
		set_variable = { rn_fp_ovl = temp_var:rn_fp_cur }
	}
	var:rn_fp_target = { rn_fp_current = yes }
	set_variable = { rn_fp_init = temp_var:rn_fp_cur }
	set_variable = { rn_fp_sel = rn_fp_init }
	set_variable = { rn_fp_letter = 1 }
	rn_fp_show_target = yes
}

# ROOT scope. Closes either picker; safe to call when none is open.
rn_fp_close = {
	if = {
		limit = { OR = { has_country_flag = rn_fp_open_own has_country_flag = rn_fp_open_sub } }
		add_to_variable = { rn_fp_epoch = 1 }
		if = { limit = { check_variable = { rn_fp_epoch > 999999 } } set_variable = { rn_fp_epoch = 1 } }
	}
	clr_country_flag = rn_fp_open_own
	clr_country_flag = rn_fp_open_sub
	clr_country_flag = rn_fp_view_target
	clr_country_flag = rn_fp_view_other
	clear_array = rn_fp_target_arr
	clear_array = rn_fp_src_arr
	clear_array = rn_fp_list
	clear_array = rn_fp_cells
	clear_array = rn_fp_rows
	clear_variable = rn_fp_target
	clear_variable = rn_fp_src
	clear_variable = rn_fp_sel
	clear_variable = rn_fp_init
	clear_variable = rn_fp_ovl
	clear_variable = rn_fp_view
	clear_variable = rn_fp_page
	clear_variable = rn_fp_pages
	clear_variable = rn_fp_page_disp
	rn_fp_touch = yes
}

# ROOT scope. The "Name" tab of the own picker: back to the country name editor (closed during the
# name cooldown, which is why the pencil opens the picker directly then).
rn_fp_to_name_editor = {
	if = {
		limit = { has_country_flag = rn_fp_open_own NOT = { rn_ip_cooldown_active_country = yes } NOT = { rn_feature_country_off = yes } }
		rn_fp_close = yes
		rn_cv1_status_clear = yes
		rn_inline_pair_open_country = yes
	}
}

# Country scope. Output temp rn_fp_cur: catalog index of the flag this country shows now. A flag
# layer this mod wrote (rn_cf_idx of the current catalog) wins; a layer written by another mod or
# the console has no index here (0); without a layer the game's own choice is resolved.
rn_fp_current = {
	rn_fp_catalog = yes
	set_temp_variable = { rn_fp_cur = 0 }
	if = {
		limit = { has_variable = capie_cf_ver has_variable = rn_cf_idx check_variable = { rn_cf_cat = temp_var:rn_fp_cat } }
		set_temp_variable = { rn_fp_cur = rn_cf_idx }
	}
	else_if = {
		limit = { NOT = { has_variable = capie_cf_ver } }
		rn_fp_resolve_game = yes
		set_temp_variable = { rn_fp_cur = temp_var:rn_fp_res }
	}
}

# ----------------------------------------------------------------------------- navigation

# ROOT scope. Shows the target's own pool (with the overlord's flag first when offered).
rn_fp_show_target = {
	if = {
		limit = { rn_fp_is_open = yes }
		clear_array = rn_fp_src_arr
		var:rn_fp_target = { ROOT = { add_to_array = { rn_fp_src_arr = PREV } } }
		set_variable = { rn_fp_src = rn_fp_target }
		set_variable = { rn_fp_view = 1 }
		set_variable = { rn_fp_page = 0 }
		clr_country_flag = rn_fp_view_other
		set_country_flag = rn_fp_view_target
		rn_fp_build = yes
	}
}

# ROOT scope. Country list of the current letter (only under the "all flags" rule).
rn_fp_show_list = {
	if = {
		limit = { rn_fp_is_open = yes has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_ALL } }
		if = { limit = { check_variable = { rn_fp_letter = 27 } } set_variable = { rn_fp_letter = 1 } }
		set_variable = { rn_fp_view = 2 }
		set_variable = { rn_fp_page = 0 }
		clr_country_flag = rn_fp_view_target
		clr_country_flag = rn_fp_view_other
		rn_fp_build = yes
	}
}

# ROOT scope. Input temp rn_fp_lt: letter 1..26, or 27 for the "Other" pool.
rn_fp_pick_letter = {
	if = {
		limit = {
			rn_fp_is_open = yes
			has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_ALL }
			check_variable = { temp_var:rn_fp_lt > 0 }
			check_variable = { temp_var:rn_fp_lt < 28 }
		}
		set_variable = { rn_fp_letter = temp_var:rn_fp_lt }
		set_variable = { rn_fp_page = 0 }
		clr_country_flag = rn_fp_view_target
		if = {
			limit = { check_variable = { rn_fp_letter = 27 } }
			set_variable = { rn_fp_view = 1 }
			set_country_flag = rn_fp_view_other
			clear_array = rn_fp_src_arr
			clear_variable = rn_fp_src
		}
		else = {
			set_variable = { rn_fp_view = 2 }
			clr_country_flag = rn_fp_view_other
		}
		rn_fp_build = yes
	}
}

# Scope: the country of a list row; ROOT is the player. Shows that country's pool.
rn_fp_pick_country = {
	if = {
		limit = {
			ROOT = { rn_fp_is_open = yes check_variable = { rn_fp_view = 2 } }
			has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_ALL }
			is_in_array = { array = ROOT.rn_fp_rows value = THIS }
		}
		ROOT = {
			clear_array = rn_fp_src_arr
			add_to_array = { rn_fp_src_arr = PREV }
			set_variable = { rn_fp_src = PREV }
			set_variable = { rn_fp_view = 1 }
			set_variable = { rn_fp_page = 0 }
			clr_country_flag = rn_fp_view_other
			clr_country_flag = rn_fp_view_target
			if = { limit = { is_in_array = { array = rn_fp_target_arr value = PREV } } set_country_flag = rn_fp_view_target }
			rn_fp_build = yes
		}
	}
}

# ROOT scope. Input temp rn_fp_pick: a catalog index shown on the current page.
rn_fp_pick_cell = {
	if = {
		limit = { rn_fp_is_open = yes is_in_array = { array = rn_fp_cells value = temp_var:rn_fp_pick } }
		set_variable = { rn_fp_sel = temp_var:rn_fp_pick }
		rn_cv1_status_clear = yes
		rn_fp_touch = yes
	}
}

# ROOT scope. Input temp rn_fp_step: -1 or 1.
rn_fp_page_step = {
	if = {
		limit = { rn_fp_is_open = yes }
		add_to_variable = { rn_fp_page = temp_var:rn_fp_step }
		rn_fp_build = yes
	}
}

# ROOT scope. Fills rn_fp_list for the current view and copies the current page (ten entries) to
# rn_fp_cells (flags) or rn_fp_rows (countries). The paging follows rn_build_char_page.
rn_fp_build = {
	rn_fp_catalog = yes
	clear_array = rn_fp_list
	if = {
		limit = { check_variable = { rn_fp_view = 2 } }
		rn_fp_fill_letter = yes
	}
	else_if = {
		limit = { has_country_flag = rn_fp_view_other }
		set_temp_variable = { rn_fp_g = temp_var:rn_fp_other }
		rn_fp_fill_group = yes
	}
	else = {
		if = {
			limit = { has_country_flag = rn_fp_view_target check_variable = { rn_fp_ovl > 0 } }
			add_to_array = { rn_fp_list = rn_fp_ovl }
		}
		set_temp_variable = { rn_fp_g = 0 }
		if = { limit = { has_variable = rn_fp_src } var:rn_fp_src = { rn_fp_group_of = yes } }
		if = { limit = { check_variable = { temp_var:rn_fp_g > 0 } } rn_fp_fill_group = yes }
	}
	set_variable = { rn_fp_pages = rn_fp_list^num }
	add_to_variable = { rn_fp_pages = 9 }
	set_temp_variable = { rn_fp_m = rn_fp_pages }
	modulo_temp_variable = { var = rn_fp_m value = 10 }
	subtract_from_variable = { rn_fp_pages = rn_fp_m }
	divide_variable = { rn_fp_pages = 10 }
	if = { limit = { check_variable = { rn_fp_pages < 1 } } set_variable = { rn_fp_pages = 1 } }
	set_temp_variable = { rn_fp_last = rn_fp_pages }
	subtract_from_temp_variable = { rn_fp_last = 1 }
	if = { limit = { check_variable = { rn_fp_page > rn_fp_last } } set_variable = { rn_fp_page = rn_fp_last } }
	if = { limit = { check_variable = { rn_fp_page < 0 } } set_variable = { rn_fp_page = 0 } }
	set_temp_variable = { rn_fp_lo = rn_fp_page }
	multiply_temp_variable = { rn_fp_lo = 10 }
	set_temp_variable = { rn_fp_hi = rn_fp_lo }
	add_to_temp_variable = { rn_fp_hi = 10 }
	clear_array = rn_fp_cells
	clear_array = rn_fp_rows
	for_each_loop = {
		array = rn_fp_list
		value = rn_fp_e
		index = rn_fp_i
		if = {
			limit = {
				NOT = { check_variable = { rn_fp_i < rn_fp_lo } }
				check_variable = { rn_fp_i < rn_fp_hi }
			}
			if = {
				limit = { check_variable = { rn_fp_view = 2 } }
				add_to_array = { rn_fp_rows = rn_fp_e }
			}
			else = { add_to_array = { rn_fp_cells = rn_fp_e } }
		}
	}
	set_variable = { rn_fp_page_disp = rn_fp_page }
	add_to_variable = { rn_fp_page_disp = 1 }
	rn_fp_touch = yes
}

# ----------------------------------------------------------------------------- Apply, Restore

# ROOT scope. Host switches (ADR-0036): temp rn_cv1_err 21 when the flag is switched off, or
# "other countries" while the picker is open for another country. Wins over every other error.
rn_fp_switch = {
	if = {
		limit = {
			OR = {
				rn_feature_flag_off = yes
				AND = { has_country_flag = rn_fp_open_sub rn_feature_subjects_off = yes }
			}
		}
		set_temp_variable = { rn_cv1_err = 21 }
	}
}

# ROOT scope. temp rn_cv1_err 20 when the picker is not open for a target the player may change.
rn_fp_rights = {
	if = {
		limit = {
			check_variable = { temp_var:rn_cv1_err = 0 }
			OR = {
				is_ai = yes
				has_country_flag = rn_open
				has_country_flag = rn_input
				NOT = { has_variable = rn_fp_target }
				NOT = { check_variable = { rn_fp_target_arr^num = 1 } }
				NOT = { rn_fp_is_open = yes }
				AND = { has_country_flag = rn_fp_open_own NOT = { var:rn_fp_target = { tag = ROOT } } }
				AND = { has_country_flag = rn_fp_open_sub NOT = { var:rn_fp_target = { rn_sub_may_edit = yes } } }
			}
		}
		set_temp_variable = { rn_cv1_err = 20 }
	}
}

# ROOT scope. Output temp rn_fp_ok = 1 when rn_fp_sel may be given to the target under the
# current rn_flag_pool rule (CONTRACT section 9: the effect decides, not the button). The host can
# change the rule while a picker is open, so this runs at Apply. Uses rn_fp_list as scratch.
rn_fp_sel_allowed = {
	rn_fp_catalog = yes
	set_temp_variable = { rn_fp_ok = 0 }
	if = {
		limit = { check_variable = { rn_fp_sel > 0 } NOT = { check_variable = { rn_fp_sel > temp_var:rn_fp_n } } }
		if = {
			limit = { has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_ALL } }
			set_temp_variable = { rn_fp_ok = 1 }
		}
		else = {
			clear_array = rn_fp_list
			var:rn_fp_target = { rn_fp_group_of = yes }
			if = { limit = { check_variable = { temp_var:rn_fp_g > 0 } } rn_fp_fill_group = yes }
			if = { limit = { is_in_array = { array = rn_fp_list value = rn_fp_sel } } set_temp_variable = { rn_fp_ok = 1 } }
			if = {
				limit = {
					check_variable = { temp_var:rn_fp_ok = 0 }
					has_game_rule = { rule = rn_flag_pool option = RN_FLAG_POOL_OWN_OVERLORD }
					var:rn_fp_target = { is_subject = yes }
				}
				var:rn_fp_target = { overlord = { rn_fp_current = yes } }
				if = { limit = { check_variable = { temp_var:rn_fp_cur = rn_fp_sel } } set_temp_variable = { rn_fp_ok = 1 } }
			}
		}
	}
}

# ROOT scope. Apply of either picker (capie_v1_ed__rn__flag / __subflag): CAPIE header (epoch,
# API level 6, capie_flag = 1), rights, the target's flag cooldown and the pool, then the carrier.
rn_fp_apply = {
	set_temp_variable = { rn_cv1_open = rn_fp_epoch }
	rn_cv1_header = yes
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } NOT = { check_variable = { temp_var:capie_cap > 5 } } }
		set_temp_variable = { rn_cv1_err = 15 }
	}
	else_if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } NOT = { check_variable = { temp_var:capie_flag = 1 } } }
		set_temp_variable = { rn_cv1_err = 16 }
	}
	rn_fp_switch = yes
	rn_fp_rights = yes
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } var:rn_fp_target = { rn_fp_cooldown_active = yes } }
		set_temp_variable = { rn_cv1_err = 11 }
	}
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } }
		rn_fp_sel_allowed = yes
		if = { limit = { NOT = { check_variable = { temp_var:rn_fp_ok = 1 } } } set_temp_variable = { rn_cv1_err = 17 } }
	}
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } }
		set_temp_variable = { rn_cf_in = rn_fp_sel }
		var:rn_fp_target = {
			rn_country_flag_carrier_set = yes
			rn_fp_cooldown_start = yes
		}
		rn_fp_close = yes
	}
	rn_cv1_status_record = yes
	if = { limit = { check_variable = { temp_var:rn_cv1_err > 0 } rn_fp_is_open = yes } rn_fp_build = yes }
}

# ROOT scope. "Restore game flag": removes the layer of the pinned target, under the same flag
# cooldown as Apply. It changes no name, colour or cosmetic tag and needs no CAPIE.
rn_fp_reset = {
	set_temp_variable = { rn_cv1_err = 0 }
	rn_fp_switch = yes
	rn_fp_rights = yes
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } var:rn_fp_target = { rn_fp_cooldown_active = yes } }
		set_temp_variable = { rn_cv1_err = 11 }
	}
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 0 } var:rn_fp_target = { has_variable = capie_cf_ver } }
		var:rn_fp_target = {
			rn_country_flag_carrier_clear = yes
			rn_fp_cooldown_start = yes
		}
		rn_fp_close = yes
	}
	rn_cv1_status_record = yes
}

# ----------------------------------------------------------------------------- daily check, cooldown

# Country scope with a flag layer (on_daily). A cosmetic tag change ends the layer (ADR-0032,
# CONTRACT 17.4), without a cooldown. Script knows only the cosmetic tags the game's files set:
#   - snapshot present and a known cosmetic tag that differs from it: reset;
#   - this mod's layer (rn_cf_idx), snapshot of a cosmetic tag, and now no known one: reset (the
#     snapshot is always a known one when this mod wrote it);
#   - this mod's layer without a snapshot (no known cosmetic tag at Apply) and now a known one: reset.
# A layer that a console or another mod wrote with an unknown cosmetic tag is left to CAPIE's gate.
rn_fp_daily_check = {
	rn_fp_cos_match = yes
	if = {
		limit = {
			OR = {
				AND = {
					has_variable = capie_cf_cos_len
					check_variable = { temp_var:rn_fp_cos_known = 1 }
					check_variable = { temp_var:rn_fp_cos_same = 0 }
				}
				AND = {
					has_variable = rn_cf_idx
					check_variable = { temp_var:rn_fp_cos_known = 0 }
					check_variable = { capie_cf_cos_len > 0 }
				}
				AND = {
					has_variable = rn_cf_idx
					check_variable = { temp_var:rn_fp_cos_known = 1 }
					NOT = { has_variable = capie_cf_cos_len }
				}
			}
		}
		rn_country_flag_carrier_clear = yes
	}
}

# Flag cooldown (scope: the country whose flag changes): its own counter, the same length as the
# name cooldown (decision 2026-10-14), started only by a successful Apply or Restore.
rn_fp_cooldown_start = {
	rn_ip_cooldown_days = yes
	set_variable = { rn_ip_cd_flag_until = global.num_days }
	add_to_variable = { rn_ip_cd_flag_until = temp_var:rn_ip_cd_days }
	set_variable = { rn_ip_cd_flag_date = global.date }
}
