# Rename Anything on the public CAPIE v1.1 editor grammar (plugin/capie/CONTRACT.md, API level 2).
# Used by the subject editors (capie_v1_ed__rn__sub*) and the native text field of the legacy rename
# window (capie_v1_ed__rn__legacy). The private rn_ip_* editors keep their frozen alias records.
#
# CAPIE adds capie_* records to the Apply of a registered editor: capie_cap (API level), capie_epoch
# (the opening epoch read from the carrier, or -1/-2/-3 as a sentinel), capie_nf and per field
# capie_f<n>_err/_at/_len/_0.._31. These adapters copy a field into the legacy temps rn_native_* /
# rn_nativeb_*, so the unchanged, hash-pinned validators (rn_inline_payload.txt,
# rn_inline_pair_payload.txt) check it and the existing copies fill the scratch buffers. Codes
# above 84 (the wider CAPIE alphabet) fail those validators with error 5: Rename Anything draws
# only its 84-code alphabet.
#
# Errors (temp rn_cv1_err, then ROOT rn_cv1_st_err for the status line):
#   0 ok; 1/4/5/6 field (validator); 10 CAPIE missing (no capie_cap >= 2); 11 cooldown;
#   12 stale opening (sentinel -1 or another epoch); 13 editor unavailable (sentinel -2/-3);
#   20 not allowed (target is not the player's subject, or the actor is AI).

# Header. Input temp rn_cv1_open = the epoch the editor stored at opening (>= 1).
rn_cv1_header = {
	set_temp_variable = { rn_cv1_err = 0 }
	set_temp_variable = { rn_cv1_field = 0 }
	if = {
		limit = { NOT = { check_variable = { temp_var:capie_cap > 1 } } }
		set_temp_variable = { rn_cv1_err = 10 }
	}
	else_if = {
		limit = { OR = { check_variable = { temp_var:capie_epoch = -2 } check_variable = { temp_var:capie_epoch = -3 } } }
		set_temp_variable = { rn_cv1_err = 13 }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { check_variable = { temp_var:rn_cv1_open > 0 } }
				NOT = { check_variable = { temp_var:capie_epoch = temp_var:rn_cv1_open } }
			}
		}
		set_temp_variable = { rn_cv1_err = 12 }
	}
}

# Reason of a field error for the status line (temp rn_cv1_why): 1 empty, 2 too long, 3 a
# character the game cannot show (CAPIE classes 3..6), 4 a letter outside Rename Anything's
# alphabet (validator code 5 on a field CAPIE accepted). Inputs: rn_cv1_err, rn_cv1_field.
rn_cv1_reason = {
	set_temp_variable = { rn_cv1_why = 0 }
	if = {
		limit = { check_variable = { temp_var:rn_cv1_field = 2 } }
		set_temp_variable = { rn_cv1_code = temp_var:capie_f2_err }
	}
	else = { set_temp_variable = { rn_cv1_code = temp_var:capie_f1_err } }
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 1 } check_variable = { temp_var:rn_cv1_code = 1 } }
		set_temp_variable = { rn_cv1_why = 1 }
	}
	else_if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 1 } check_variable = { temp_var:rn_cv1_code = 2 } }
		set_temp_variable = { rn_cv1_why = 2 }
	}
	else_if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 1 } check_variable = { temp_var:rn_cv1_code > 2 } check_variable = { temp_var:rn_cv1_code < 7 } }
		set_temp_variable = { rn_cv1_why = 3 }
	}
	else_if = {
		limit = { check_variable = { temp_var:rn_cv1_err = 5 } }
		set_temp_variable = { rn_cv1_why = 4 }
	}
}

# Status line (display only, ROOT): written after every Apply of a grammar editor, cleared on open.
rn_cv1_status_clear = {
	clear_variable = rn_cv1_st_err
	clear_variable = rn_cv1_st_field
	clear_variable = rn_cv1_st_why
}
rn_cv1_status_record = {
	rn_cv1_status_clear = yes
	if = {
		limit = { check_variable = { temp_var:rn_cv1_err > 0 } }
		rn_cv1_reason = yes
		set_variable = { rn_cv1_st_err = temp_var:rn_cv1_err }
		set_variable = { rn_cv1_st_field = temp_var:rn_cv1_field }
		if = {
			limit = { check_variable = { temp_var:rn_cv1_why > 0 } }
			set_variable = { rn_cv1_st_why = temp_var:rn_cv1_why }
		}
	}
}

# One field (field 1 -> rn_native_*). Output rn_cv1_err/rn_cv1_field; no persistent write.
rn_cv1_take_one = {
	rn_cv1_take_1_as_a = yes
	rn_inline_validate_payload = yes
	if = {
		limit = { NOT = { check_variable = { temp_var:rn_inline_error = 0 } } }
		set_temp_variable = { rn_cv1_err = temp_var:rn_inline_error }
		set_temp_variable = { rn_cv1_field = 1 }
	}
}

# Two fields (1 -> rn_native_*, 2 -> rn_nativeb_*). Output rn_cv1_err/rn_cv1_field; no persistent write.
rn_cv1_take_two = {

	rn_inline_pair_validate = yes
	if = {
		limit = { NOT = { check_variable = { temp_var:rn_pair_err = 0 } } }
		set_temp_variable = { rn_cv1_err = temp_var:rn_pair_err }
		set_temp_variable = { rn_cv1_field = temp_var:rn_pair_field }
	}
}
