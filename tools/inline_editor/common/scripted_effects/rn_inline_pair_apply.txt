# Rename Anything in-place editor. No foreign-country/MP authority claim.
# player_context is required; tag=ROOT verifies CURRENT==ROOT, not local authority by itself.
# Party logical group1/2/3/4 = democratic/communism/fascism/neutrality, not engine IDs.
# Full pure context/epoch/typed-country/group/cap guard -> strict pair validation ->
# COMPLETE guard repeated before the first persistent Apply write.
# Four FIXED literal setter branches, both fields expanded by meta_effect.
# Error7 context,8 epoch,9 pin/current group,10 fresh country capability; payload1/4/5/6.
# Validation-before-writes is not rollback against an engine failure or crash.
# Government mismatch preserves draft/open/pins/epoch and buffers; Cancel remains available.
rn_inline_pair_guard_country = {
	set_temp_variable = { rn_pair_err = 7 }
	set_temp_variable = { rn_pair_field = 0 }
	set_temp_variable = { rn_pair_bad = -1 }
	if = {
		limit = {
			tag = ROOT
			is_ai = no
			NOT = { has_country_flag = rn_open }
			NOT = { has_country_flag = rn_input }
			NOT = { has_country_flag = rn_ip_leader_open }
			has_country_flag = rn_ip_country_open
			NOT = { has_country_flag = rn_ip_party_neutrality_open }
		}
		set_temp_variable = { rn_pair_err = 8 }
		set_temp_variable = { rn_pair_epoch_round = rn_ip_epoch }
		round_temp_variable = rn_pair_epoch_round
		if = {
			limit = {
				check_variable = { rn_ip_epoch > 0 }
				check_variable = { temp_var:rn_pair_epoch_round = rn_ip_epoch }
				check_variable = { temp_var:rn_ip_cmd_epoch = rn_ip_epoch }
			}
			set_temp_variable = { rn_pair_err = 9 }
			if = {
				limit = {
					check_variable = { rn_ip_country_target^num = 1 }
					is_in_array = { array = rn_ip_country_target value = THIS }
				}
				set_temp_variable = { rn_pair_err = 10 }
				if = {
					limit = { check_variable = { temp_var:rn_native_cap = 1 } }
					set_temp_variable = { rn_pair_err = 0 }
				}
			}
		}
	}
	# Iteration 3: per-type rename cooldown (error 11), re-checked by BOTH guard rounds.
	# Host switch (ADR-0036): error 21, before the cooldown.
	if = {
		limit = { check_variable = { temp_var:rn_pair_err = 0 } rn_feature_country_off = yes }
		set_temp_variable = { rn_pair_err = 21 }
	}
	if = {
		limit = { check_variable = { temp_var:rn_pair_err = 0 } rn_ip_cooldown_active_country = yes }
		set_temp_variable = { rn_pair_err = 11 }
	}
}

rn_inline_pair_guard_party = {
	set_temp_variable = { rn_pair_err = 7 }
	set_temp_variable = { rn_pair_field = 0 }
	set_temp_variable = { rn_pair_bad = -1 }
	if = {
		limit = {
			tag = ROOT

			NOT = { has_country_flag = rn_open }
			NOT = { has_country_flag = rn_input }
			NOT = { has_country_flag = rn_ip_leader_open }
			NOT = { has_country_flag = rn_ip_country_open }
			has_country_flag = rn_ip_party_neutrality_open
		}
		set_temp_variable = { rn_pair_err = 8 }
		set_temp_variable = { rn_pair_epoch_round = rn_ip_epoch }
		round_temp_variable = rn_pair_epoch_round
		if = {
			limit = {
				check_variable = { rn_ip_epoch > 0 }
				check_variable = { temp_var:rn_pair_epoch_round = rn_ip_epoch }
				check_variable = { temp_var:rn_ip_cmd_epoch = rn_ip_epoch }
			}
			set_temp_variable = { rn_pair_err = 9 }
			if = {
				limit = {
					check_variable = { rn_ip_party_target^num = 1 }
					is_in_array = { array = rn_ip_party_target value = THIS }
					OR = { check_variable = { rn_ip_party_group = 1 } check_variable = { rn_ip_party_group = 2 } check_variable = { rn_ip_party_group = 3 } check_variable = { rn_ip_party_group = 4 } }
				}
				set_temp_variable = { rn_pair_err = 0 }
			}
		}
	}
	# Iteration 3: per-type rename cooldown (error 11), re-checked by BOTH guard rounds.
	# Host switch (ADR-0036): error 21, before the cooldown.
	if = {
		limit = { check_variable = { temp_var:rn_pair_err = 0 } rn_feature_party_off = yes }
		set_temp_variable = { rn_pair_err = 21 }
	}
	if = {
		limit = { check_variable = { temp_var:rn_pair_err = 0 } rn_ip_cooldown_active_party = yes }
		set_temp_variable = { rn_pair_err = 11 }
	}
}

rn_inline_pair_open_party = {
 if = {
  limit = {
   tag = ROOT is_ai = no
   NOT = { has_country_flag = rn_open }
   NOT = { has_country_flag = rn_input }
   NOT = { has_country_flag = rn_ip_leader_open }
   NOT = { has_country_flag = rn_ip_country_open }
   NOT = { has_country_flag = rn_ip_party_neutrality_open }
   NOT = { rn_ip_cooldown_active_party = yes } NOT = { rn_feature_party_off = yes }
   OR = { has_government = democratic has_government = communism has_government = fascism has_government = neutrality }
  }
  if = {
   limit = { has_government = democratic }
   clear_array = rn_ip_party_target
   add_to_array = { rn_ip_party_target = THIS }
   set_variable = { rn_ip_party_group = 1 }
   add_to_variable = { rn_ip_epoch = 1 }
   set_country_flag = rn_ip_party_neutrality_open
  }
  else_if = {
   limit = { has_government = communism }
   clear_array = rn_ip_party_target
   add_to_array = { rn_ip_party_target = THIS }
   set_variable = { rn_ip_party_group = 2 }
   add_to_variable = { rn_ip_epoch = 1 }
   set_country_flag = rn_ip_party_neutrality_open
  }
  else_if = {
   limit = { has_government = fascism }
   clear_array = rn_ip_party_target
   add_to_array = { rn_ip_party_target = THIS }
   set_variable = { rn_ip_party_group = 3 }
   add_to_variable = { rn_ip_epoch = 1 }
   set_country_flag = rn_ip_party_neutrality_open
  }
  else_if = {
   limit = { has_government = neutrality }
   clear_array = rn_ip_party_target
   add_to_array = { rn_ip_party_target = THIS }
   set_variable = { rn_ip_party_group = 4 }
   add_to_variable = { rn_ip_epoch = 1 }
   set_country_flag = rn_ip_party_neutrality_open
  }
 }
}
