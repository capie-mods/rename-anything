# Country colour carrier: the only place that writes the CAPIE colour layer (plugin/capie/CONTRACT.md
# section 19, API level 8, CAPIE 1.6). Scope: the country whose map colour changes.
#   capie_cc_r / capie_cc_g / capie_cc_b  the colour, integers 0..255 as in colors.txt rgb { }; CAPIE
#                                        applies the game's country colour defines to it
#   capie_cc_cos / capie_cc_cos_len       snapshot of the cosmetic tag at the time of the choice
#                                        (as in 17.2 and 17.3); absent when the country has no
#                                        cosmetic tag this mod knows
#   capie_cc_ver = 1                      layer set (format 1); capie_cc_rev = 1 always LAST
# Mod bookkeeping next to it: rn_cc_mine (this mod wrote the layer; the daily cosmetic check).
# The name layer, the flag layer and the cosmetic tag are never touched here.

# Input temps rn_cc_r, rn_cc_g, rn_cc_b: the colour. Each channel is rounded and clamped to 0..255
# first, so the layer is never invalid.
rn_country_colour_carrier_set = {
	round_temp_variable = rn_cc_r
	round_temp_variable = rn_cc_g
	round_temp_variable = rn_cc_b
	clamp_temp_variable = { var = rn_cc_r min = 0 max = 255 }
	clamp_temp_variable = { var = rn_cc_g min = 0 max = 255 }
	clamp_temp_variable = { var = rn_cc_b min = 0 max = 255 }
	set_variable = { capie_cc_r = temp_var:rn_cc_r }
	set_variable = { capie_cc_g = temp_var:rn_cc_g }
	set_variable = { capie_cc_b = temp_var:rn_cc_b }
	rn_cp_cos_write = yes
	set_variable = { rn_cc_mine = 1 }
	set_variable = { capie_cc_ver = 1 }
	set_variable = { capie_cc_rev = 1 }
}

# Removes the layer; the game's colour (cosmetic tag colour, else base colour) comes back.
rn_country_colour_carrier_clear = {
	clear_variable = capie_cc_r
	clear_variable = capie_cc_g
	clear_variable = capie_cc_b
	clear_array = capie_cc_cos
	clear_variable = capie_cc_cos_len
	clear_variable = capie_cc_ver
	clear_variable = rn_cc_mine
	set_variable = { capie_cc_rev = 1 }
}
