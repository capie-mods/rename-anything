# Flag picker triggers (see common/scripted_effects/rn_flag_picker.txt).

# ROOT (player) scope: a flag picker is open and nothing else blocks it.
rn_fp_is_open = {
	is_ai = no
	NOT = { has_country_flag = rn_open }
	NOT = { has_country_flag = rn_input }
	OR = { has_country_flag = rn_fp_open_own has_country_flag = rn_fp_open_sub }
}

# Scope: the country whose flag changes. Its flag cooldown is still running.
rn_fp_cooldown_active = {
	set_temp_variable = { rn_ip_cd_left = rn_ip_cd_flag_until }
	subtract_from_temp_variable = { rn_ip_cd_left = global.num_days }
	check_variable = { rn_ip_cd_left > 0 }
}
