# Rename rights (game rule rn_edit_rights). Own country: always (its own editors keep their guards).
#   RN_RIGHTS_SUBJECTS (default)  own country and own subjects; faction: its leader
#   RN_RIGHTS_HOST                as above, and the host renames every country and every faction
#   RN_RIGHTS_OWN                 own country only; no subjects; faction: its leader (the faction is
#                                 the leader's own)
# The host: the only human in single player, or the country whose player sent a CAPIE command with
# the record capie_host = 1 (rn_cv1_header sets the flag rn_is_host). The game itself has no host
# trigger (docs/API-DECISIONS.md, 2026-10-11 correction).

# ROOT scope: the acting player is the host.
rn_rights_host = {
	is_ai = no
	OR = {
		NOT = { any_other_country = { is_ai = no } }
		has_country_flag = rn_is_host
	}
}

# Another country's names (scope: the target country, ROOT: the actor).
rn_sub_may_edit = {
	exists = yes
	NOT = { tag = ROOT }
	OR = {
		AND = {
			NOT = { has_game_rule = { rule = rn_edit_rights option = RN_RIGHTS_OWN } }
			is_subject_of = ROOT
		}
		AND = {
			has_game_rule = { rule = rn_edit_rights option = RN_RIGHTS_HOST }
			ROOT = { rn_rights_host = yes }
		}
	}
}

# A faction's name (scope: the faction leader, ROOT: the actor).
rn_fac_may_edit = {
	exists = yes
	is_faction_leader = yes
	OR = {
		tag = ROOT
		AND = {
			has_game_rule = { rule = rn_edit_rights option = RN_RIGHTS_HOST }
			ROOT = { rn_rights_host = yes }
		}
	}
}

# Faction rename cooldown (scope: the faction leader), the same length as every other rename.
rn_fac_cooldown_active = {
	set_temp_variable = { rn_ip_cd_left = rn_fac_cd_until }
	subtract_from_temp_variable = { rn_ip_cd_left = global.num_days }
	check_variable = { rn_ip_cd_left > 0 }
}
