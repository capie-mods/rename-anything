		context_type = player_context
		parent_window_token = politics_tab
		window_name = "rn_ip_country_launcher_window"
		visible = { tag = ROOT is_ai = no rn_editor_bridge_available = yes NOT = { has_country_flag = rn_ip_country_open } NOT = { has_country_flag = rn_fp_open_own } NOT = { has_country_flag = rn_open } NOT = { has_country_flag = rn_input } }

					rn_ip_status_clear = yes

			rn_ip_country_open_click_enabled = {

				is_ai = no
				NOT = { has_country_flag = rn_ip_country_open }
				NOT = { has_country_flag = rn_ip_leader_open }
				NOT = { has_country_flag = rn_ip_party_neutrality_open }
				NOT = { has_country_flag = rn_open }
				NOT = { has_country_flag = rn_input }

		context_type = player_context
		parent_window_token = politics_tab
		window_name = "rn_ip_country_window"
		visible = { tag = ROOT is_ai = no rn_editor_bridge_available = yes has_country_flag = rn_ip_country_open NOT = { has_country_flag = rn_open } NOT = { has_country_flag = rn_input } }

			# Actual guarded Apply: enabled checks own typed pin, while ctor supplies fresh payload/cap.
			rn_ip_country_apply_click = {
				log = "RN_IPP country before name=[ROOT.GetName] adjective=[ROOT.GetAdjective] cap=[?temp_var:rn_native_cap] cmd_epoch=[?temp_var:rn_ip_cmd_epoch|0] epoch=[?ROOT.rn_ip_epoch|0] pin_count=[?ROOT.rn_ip_country_target^num|0]"

				rn_ip_status_record_pair = yes
				log = "RN_IPP country after err=[?temp_var:rn_pair_err] field=[?temp_var:rn_pair_field] bad=[?temp_var:rn_pair_bad] name=[ROOT.GetName] adjective=[ROOT.GetAdjective] cap=[?temp_var:rn_native_cap] cmd_epoch=[?temp_var:rn_ip_cmd_epoch|0] epoch=[?ROOT.rn_ip_epoch|0] pin_count=[?ROOT.rn_ip_country_target^num|0]"

				rn_inline_pair_cancel_country = yes
				rn_ip_status_clear = yes
			}
			# Restore the game's name: the same own-country checks and the same cooldown as Apply.
			rn_ip_country_reset_click = {
				rn_inline_country_reset = yes
				rn_ip_status_clear = yes
			}
			# Tabs: Name is this editor; Flag opens the flag picker in the same place.
			rn_ip_country_tab_flag_click = {
				rn_fp_open_own = yes
				rn_ip_status_clear = yes
			}
		}
		triggers = {
			rn_ip_country_name_prefill_visible = { always = no }

				is_ai = no
				has_country_flag = rn_ip_country_open
				NOT = { has_country_flag = rn_ip_leader_open }
				NOT = { has_country_flag = rn_ip_party_neutrality_open }
				NOT = { has_country_flag = rn_open }
				NOT = { has_country_flag = rn_input }

				check_variable = { rn_ip_country_target^num = 1 }

			rn_ip_country_tab_name_click_enabled = { tag = ROOT is_ai = no always = no }

		}
	}
